# ---------- 5 System design
def big(s, x, y, icon, title, sub, accent=INK, w=1.55, h=1.12):
    box(s, x, y, w, h, WHITE, accent, lw=1.25)
    isz = .44
    s.shapes.add_picture("midreview_assets/icons/%s.png" % icon, Inches(x + w / 2 - isz / 2), Inches(y + .08), width=Inches(isz))
    text(s, x + .02, y + .56, w - .04, .26, title, 11.5, accent, True, PP_ALIGN.CENTER, font=SERIF)
    text(s, x + .02, y + .81, w - .04, .24, sub, 9.5, MUTED, False, PP_ALIGN.CENTER)


s = new_slide()
header(s, 6, "System design")

big(s, CXL[1], yA, "snap_ink", "Map matching", "GPS to road edges", INK)
big(s, CXL[2], yA, "route_ink", "Expert routes", "80% train, 20% test", INK)
big(s, CXL[3], yA, "cycle", "MaxEnt IRL", "learns the reward", ACC)
big(s, CXL[4], yA, "db_acc", "Weights w*", "saved model", ACC)

    arrow(s, CXL[i] + 1.55, yA + .56, CXL[i + 1], yA + .56, MUTED, w=1.6)
