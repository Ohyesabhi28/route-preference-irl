def node(s, x, y, w, h, icon, title, sub, accent=INK, dash=False):
    box(s, x, y, w, h, WHITE, accent, lw=1, dash=dash)
    isz = .36
    s.shapes.add_picture("midreview_assets/icons/%s.png" % icon, Inches(x + w / 2 - isz / 2), Inches(y + .07), width=Inches(isz))
    text(s, x + .03, y + .46, w - .06, .24, title, 10, accent, True, PP_ALIGN.CENTER, font=SERIF)
    text(s, x + .03, y + .7, w - .06, .22, sub, 8.5, MUTED, False, PP_ALIGN.CENTER)


def band(s, y, h, label, fill, edge, x=.62):
    box(s, .5, y, 9, h, fill, edge, lw=.75, dash=True)
    text(s, x, y + .02, 5.6, .26, label, 11, edge, True, font=SERIF)
