def mini(s, x, y, w, h, icon, title, sub, accent=INK, dash=False, fill=WHITE):
    box(s, x, y, w, h, fill, accent, lw=1, dash=dash)
    isz = .28
    s.shapes.add_picture("midreview_assets/icons/%s.png" % icon, Inches(x + w / 2 - isz / 2), Inches(y + .04), width=Inches(isz))
    text(s, x + .02, y + .32, w - .04, .22, title, 9, accent, True, PP_ALIGN.CENTER, font=SERIF)
    if sub:
